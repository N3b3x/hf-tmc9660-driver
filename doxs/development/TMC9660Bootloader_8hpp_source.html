1342static constexpr uint8_t crc8Bootloader(const uint8_t* data, size_t len) noexcept {
1343 uint16_t crc = 0; // 9-bit register for polynomial division
1344 const uint16_t POLY = 0x107; // x^8 + x^2 + x^1 + x^0 in 9-bit form
1345
1346 // Process each byte (bit-reversed, LSB-first)
1347 for (size_t i = 0; i < len; i++) {
1348 uint8_t byte_reversed = reverseByte(data[i]);
1349
1350 // Feed 8 bits into CRC register (MSB-first from reversed byte)
1351 for (int bit = 7; bit >= 0; bit--) {
1352 crc = (crc << 1) | ((byte_reversed >> bit) & 1);
1353 if (crc & 0x100) { // If bit 8 is set
1354 crc ^= POLY;
1355 }
1356 }
1357 }
1358
1359 // Append 8 zero bits (flush remaining data through CRC)
1360 for (int i = 0; i < 8; i++) {
1361 crc <<= 1;
1362 if (crc & 0x100) {
1363 crc ^= POLY;
1364 }
1365 }
1366
1367 return static_cast<uint8_t>(crc & 0xFF);
1368}

1472 // (No SYNC byte in reply, unlike the request which has 0x55 as first byte)
1473 r.hostAddr = in[0];
1474 r.deviceAddr = in[1];
1475 r.status = in[2];

1477 (static_cast<uint32_t>(in[4]) << 16) |
1478 (static_cast<uint32_t>(in[5]) << 8) |
1479 static_cast<uint32_t>(in[6]);
1480 // Note: in[7] is CRC8, should be verified
1481 return r;
1482 }
