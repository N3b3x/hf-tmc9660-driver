542 bool transferTMCL(const TMCLFrame &tx, TMCLReply &reply, uint8_t address,
543 TMCLReply *firstReply, const TMCLFrame *secondCommand) noexcept override {
544 // UART doesn't use the two-transaction pattern, so firstReply and secondCommand are ignored
545 (void)firstReply; // Suppress unused parameter warning
546 (void)secondCommand; // Suppress unused parameter warning
547 if (!initialized_) {
548 ESP_LOGE(BUS_TAG, "UART interface not initialized");
549 return false;
550 }
551
552 // Clear RX buffer before sending
553 uart_flush_input(uart_num_);
554
555 // ✅ FIX: Use TMCLFrame::toUart() for correct frame encoding
556 // Previous manual packing had multiple bugs:
557 // - Wrong sync bit position (should be bit 0, not 0x55 in byte 0)
558 // - Only wrote lower 8 bits of 16-bit type field
559 // - Array indexing off by one (overwrote checksum with value LSB)
560 std::array<uint8_t, 9> uart_frame;
561 tx.toUart(address, uart_frame); // Correctly encodes all fields + checksum
562
563 // Send 9-byte UART frame
564 int bytes_written = uart_write_bytes(uart_num_, uart_frame.data(), uart_frame.size());
565 if (bytes_written != static_cast<int>(uart_frame.size())) {
566 ESP_LOGE(BUS_TAG, "UART write failed: expected %zu, wrote %d", uart_frame.size(), bytes_written);
567 return false;
568 }
569
570 // Log transmitted bytes
571 logDebug(2, "UART_TMCL", "[UART TX] %02X %02X %02X %02X %02X %02X %02X %02X %02X",
572 uart_frame[0], uart_frame[1], uart_frame[2], uart_frame[3],
573 uart_frame[4], uart_frame[5], uart_frame[6], uart_frame[7], uart_frame[8]);
574
575 // Wait for transmission to complete
576 uart_wait_tx_done(uart_num_, portMAX_DELAY);
577
578 // Receive 9-byte UART reply
579 std::array<uint8_t, 9> uart_reply;
580 int bytes_read = uart_read_bytes(uart_num_, uart_reply.data(), uart_reply.size(), pdMS_TO_TICKS(10));
581 if (bytes_read != static_cast<int>(uart_reply.size())) {
582 ESP_LOGE(BUS_TAG, "UART read failed: expected %zu, read %d", uart_reply.size(), bytes_read);
583 // Log partial data if any was received
584 if (bytes_read > 0) {
585 logDebug(2, "UART_TMCL", "[UART RX] (partial %d bytes) %02X %02X %02X %02X %02X %02X %02X %02X %02X",
586 bytes_read,
587 bytes_read > 0 ? uart_reply[0] : 0, bytes_read > 1 ? uart_reply[1] : 0,
588 bytes_read > 2 ? uart_reply[2] : 0, bytes_read > 3 ? uart_reply[3] : 0,
589 bytes_read > 4 ? uart_reply[4] : 0, bytes_read > 5 ? uart_reply[5] : 0,
590 bytes_read > 6 ? uart_reply[6] : 0, bytes_read > 7 ? uart_reply[7] : 0,
591 bytes_read > 8 ? uart_reply[8] : 0);
592 }
593 return false;
594 }
595
596 // Log received bytes
597 logDebug(2, "UART_TMCL", "[UART RX] %02X %02X %02X %02X %02X %02X %02X %02X %02X",
598 uart_reply[0], uart_reply[1], uart_reply[2], uart_reply[3],
599 uart_reply[4], uart_reply[5], uart_reply[6], uart_reply[7], uart_reply[8]);
600
601 // Use TMCLReply::fromUart() for correct decoding with command context
602 // This handles checksum verification and proper field extraction
603 // Pass command context (opcode, type) for handling special reply formats like GetVersion string
604 if (!TMCLReply::fromUart(uart_reply, address, reply, tx.opcode, tx.type)) {
605 ESP_LOGE(BUS_TAG, "Failed to parse UART reply (checksum or address mismatch)");
606 return false;
607 }
608
609 return true;
610 }

775 input_config.pull_up_en = GPIO_PULLUP_ENABLE; // Enable pull-up for open-drain FAULTN
776

780 return false;
781 }

784 gpio_set_level(rst_pin_, 0); // RST active HIGH, start with inactive (low)
785 gpio_set_level(drv_en_pin_, 0); // DRV_EN active HIGH, start with disabled (low)
786 gpio_set_level(wake_pin_, 1); // WAKE active LOW, start with inactive (high)
787
788 return true;
789 }
