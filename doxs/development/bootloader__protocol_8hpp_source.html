344struct BootloaderCommandUART {
345 uint8_t deviceAddr;
346 uint8_t command;
347 uint32_t value;
348
350 void toBuffer(std::array<uint8_t, 8>& out) const noexcept {
351 out[0] = 0x55; // Sync byte
352 out[1] = deviceAddr;
353 out[2] = command;
354 out[3] = static_cast<uint8_t>(value >> 24); // MSB first
355 out[4] = static_cast<uint8_t>(value >> 16);
356 out[5] = static_cast<uint8_t>(value >> 8);
357 out[6] = static_cast<uint8_t>(value);
358 out[7] = crc8Bootloader(out.data(), 7); // CRC over first 7 bytes
359 }

406struct BootloaderReplyUART {
407 uint8_t hostAddr;
408 uint8_t deviceAddr;
409 uint8_t status;
410 uint32_t value;
411
413 static BootloaderReplyUART fromBuffer(const std::array<uint8_t, 8>& in) noexcept {
414 BootloaderReplyUART r;
415 // UART protocol: [HOST_ADDR] [DEVICE_ADDR] [STATUS] [VALUE(32)] [CRC8]
416 // (No SYNC byte in reply, unlike the request which has 0x55 as first byte)
417 r.hostAddr = in[0];
418 r.deviceAddr = in[1];
419 r.status = in[2];
420 r.value = (static_cast<uint32_t>(in[3]) << 24) | // MSB first
421 (static_cast<uint32_t>(in[4]) << 16) | (static_cast<uint32_t>(in[5]) << 8) |
422 static_cast<uint32_t>(in[6]);
423 // Note: in[7] is CRC8, should be verified
424 return r;
425 }
426
428 bool verifyCRC(const std::array<uint8_t, 8>& in) const noexcept {
429 uint8_t calculatedCRC = crc8Bootloader(in.data(), 7);
430 return calculatedCRC == in[7];
431 }
432
434 bool isOK() const noexcept {
435 return status == static_cast<uint8_t>(BootloaderStatus::OK);
436 }
437
439 BootloaderStatus getStatus() const noexcept {
440 return static_cast<BootloaderStatus>(status);
441 }
