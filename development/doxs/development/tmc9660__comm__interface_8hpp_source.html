289 static bool fromSpi(std::span<const uint8_t, 8> in, TMCLReply& r, uint8_t sent_opcode = 0,
290 uint16_t sent_type = 0) noexcept {
291 // Store raw bytes for advanced parsing (ALWAYS, even if parsing fails)
292 std::copy(in.begin(), in.end(), r.rawBytes.begin());
293
294 // Check for SESSION_START status codes (bootloader or parameter mode)
295 uint8_t raw_status = in[0];
296 if (raw_status == 0x13) {
297 // SESSION_START from bootloader (0x13)
298 // These use different protocols, so TMCL checksum will fail
299 // But we still want to return success so caller can use rawBytes
300 r.spi_status = static_cast<SPIStatus>(raw_status);
301 r.status = in[1];
302 r.opcode = in[2];
303 r.value = (static_cast<uint32_t>(in[3]) << 24) | (static_cast<uint32_t>(in[4]) << 16) |
304 (static_cast<uint32_t>(in[5]) << 8) | static_cast<uint32_t>(in[6]);
305 return true; // Allow SESSION_START even if checksum fails
306 }
307
308 // Check for GetVersion string format (Command 136, Type 0)
309 // GetVersion with Type=0 returns a special string format without checksum
310 // Format: [Host Address][Version String 8 chars]
311 if (sent_opcode == 136 && sent_type == 0) {
312 // We sent GetVersion Type=0, expect string format reply (no checksum validation)
313 r.spi_status = SPIStatus::OK; // Assume OK for version string
314 r.status = 100; // REPLY_OK for successful version retrieval
315 r.opcode = 136; // GetVersion command
316 r.value = 0; // No value field for string format
317 return true; // Success - skip checksum validation for string format
318 }
319
320 // Standard TMCL reply validation
321 if (tmclChecksum(in.data(), 7) != in[7])
322 return false;
323 r.spi_status = static_cast<SPIStatus>(in[0]);
324 if (r.spi_status == SPIStatus::NOT_READY || r.spi_status == SPIStatus::CHECKSUM_ERROR)
325 return false;
326 r.status = in[1];
327 r.opcode = in[2];
328 r.value = (static_cast<uint32_t>(in[3]) << 24) | (static_cast<uint32_t>(in[4]) << 16) |
329 (static_cast<uint32_t>(in[5]) << 8) | static_cast<uint32_t>(in[6]);
330 return true;
331 }

350 static bool fromUart(std::span<const uint8_t, 9> in, uint8_t addr, TMCLReply& r,
351 uint8_t sent_opcode = 0, uint16_t sent_type = 0) noexcept {
352 // Store raw bytes (map 9-byte UART to 8-byte format, skip first byte)
353 std::copy(in.begin() + 1, in.end(), r.rawBytes.begin());
354
355 // UART Reply Format (per datasheet):
356 // byte0: Host Address (8 bits)
357 // byte1: Sync bit (bit 0) + Module Address (bits 1-7)
358 // byte2: TMCL Status
359 // byte3: Operation
360 // byte4-7: Data (big-endian)
361 // byte8: Checksum
362
363 // Check for GetVersion string format (Command 136, Type 0)
364 // GetVersion with Type=0 returns a special string format without valid checksum
365 // Format: [Host Address][Sync+Address][Version String 7 chars]
366 if (sent_opcode == 136 && sent_type == 0) {
367 // We sent GetVersion Type=0, expect string format reply (no checksum validation)
368 r.spi_status = SPIStatus::OK; // Assume OK for version string
369 r.status = 100; // REPLY_OK for successful version retrieval
370 r.opcode = 136; // GetVersion command
371 r.value = 0; // No value field for string format
372 return true; // Success - skip checksum validation for string format
373 }
374
375 // Verify module address in byte 1 (bits 1-7)
376 // "The module address reuses the upper 7 bits of the bootloader device address"
377 // Compare the upper 7 bits of byte 1 with the upper 7 bits of the address
378 if ((in[1] & 0xFE) != (addr & 0xFE))
379 return false;
380
381 // Verify checksum over first 8 bytes
382 if (tmclChecksum(in.data(), 8) != in[8])
383 return false;
384
385 r.spi_status = SPIStatus::OK; // UART has no SPI status field
386 r.status = in[2]; // TMCL Status at byte 2
387 r.opcode = in[3]; // Operation at byte 3
388 r.value = (static_cast<uint32_t>(in[4]) << 24) | (static_cast<uint32_t>(in[5]) << 16) |
389 (static_cast<uint32_t>(in[6]) << 8) | static_cast<uint32_t>(in[7]);
390 return true;
391 }

399struct TMCLFrame {
400 uint8_t opcode = 0;
401 uint16_t type = 0;

409 void toSpi(std::span<uint8_t, 8> out) const noexcept {
410 out[0] = opcode;
411 out[1] = static_cast<uint8_t>(type & 0xFF); // Type lower 8 bits (bits 0-7)
412 // BYTE2: bits 20-23 (high nibble) = Motor/Bank, bits 16-19 (upper nibble) = Type upper 4 bits
413 out[2] = static_cast<uint8_t>(((motor & 0x0F) << 4) | ((type & 0xF00) >> 8));
414 out[3] = static_cast<uint8_t>(value >> 24);
415 out[4] = static_cast<uint8_t>(value >> 16);
416 out[5] = static_cast<uint8_t>(value >> 8);
417 out[6] = static_cast<uint8_t>(value);
418 out[7] = tmclChecksum(out.data(), 7);
419 }
420
426 void toUart(uint8_t addr, std::span<uint8_t, 9> out) const noexcept {
427 // Byte 0: [Sync bit (bit 0)] [Module Address (bits 1-7)]
428 // "The module address reuses the upper 7 bits of the bootloader device address"
429 // This means we take bits 7-1 of the address and place them in bits 7-1 of byte 0
430 // Then set bit 0 (sync bit) to 1
431 out[0] = (addr & 0xFE) | 0x01; // Keep upper 7 bits of address, set sync bit
432 out[1] = opcode;
433 out[2] = static_cast<uint8_t>(type & 0xFF); // Type field (8 bits)
434 out[3] = static_cast<uint8_t>(motor & 0x0F); // Motor/Bank field (4 bits)
435 out[4] = static_cast<uint8_t>(value >> 24);
436 out[5] = static_cast<uint8_t>(value >> 16);
437 out[6] = static_cast<uint8_t>(value >> 8);
438 out[7] = static_cast<uint8_t>(value);
439 out[8] = tmclChecksum(out.data(), 8);
440 }

447 static TMCLFrame fromSpi(std::span<const uint8_t, 8> in) noexcept {
448 TMCLFrame f;
449 f.opcode = in[0];
450 // BYTE1: Type lower 8 bits (bits 0-7)
451 // BYTE2: bits 20-23 (high nibble) = Motor/Bank, bits 16-19 (low nibble) = Type upper 4 bits
452 f.type = static_cast<uint16_t>(in[1] | ((in[2] & 0x0F) << 8));
453 f.motor = (in[2] >> 4) & 0x0F;
454 f.value = (static_cast<uint32_t>(in[3]) << 24) | (static_cast<uint32_t>(in[4]) << 16) |
455 (static_cast<uint32_t>(in[5]) << 8) | static_cast<uint32_t>(in[6]);
456 return f;

955 const TMCLFrame* second_command) noexcept {
956 std::array<uint8_t, 8> tx_buf, rx_buf;
957 tx.toSpi(tx_buf);
958
959 // Log raw SPI bytes being transmitted
960 TMC9660_LOG_DEBUG(*static_cast<Derived*>(this), 2, "SPI_TMCL", "[TMCL TX 1 ] %02X %02X %02X %02X %02X %02X %02X %02X",
961 tx_buf[0], tx_buf[1], tx_buf[2], tx_buf[3], tx_buf[4], tx_buf[5], tx_buf[6],
962 tx_buf[7]);
963
964 // Transaction 1: Send command, receive first reply
965 if (!spiTransferTMCL(tx_buf, rx_buf))
966 return false;
967
968 TMC9660_LOG_DEBUG(*static_cast<Derived*>(this), 2, "SPI_TMCL", "[TMCL RX 1 ] %02X %02X %02X %02X %02X %02X %02X %02X",
969 rx_buf[0], rx_buf[1], rx_buf[2], rx_buf[3], rx_buf[4], rx_buf[5], rx_buf[6],
970 rx_buf[7]);
971
972 // Optionally capture first reply (this is the reply to the PREVIOUS command due to SPI delay)
973 if (first_reply) {
974 bool first_parse_ok = TMCLReply::fromSpi(rx_buf, *first_reply);
975 TMC9660_LOG_DEBUG(
976 *static_cast<Derived*>(this), 2, "SPI_TMCL",
977 " └─> SPI_Status=0x%02X, TMCL_Status=0x%02X, Value=0x%08X (parse %s)",
978 static_cast<uint8_t>(first_reply->spi_status), first_reply->status, first_reply->value,
979 first_parse_ok ? "OK" : "failed");
980 // Note: We don't return false here because SESSION_START may have non-standard format
981 // The caller can check rawBytes[0] for SESSION_START status codes
982 }
983
984 // Transaction 2: Send second command (or NO_OP), receive final reply
985 // This transaction is wrapped in retry logic for SPI_STATUS_NOT_READY
986 TMCLFrame cmd2 = second_command ? *second_command : TMCLFrame{}; // NO_OP if not provided
987 cmd2.toSpi(tx_buf);
988
989 // Retry loop for SPI_STATUS_NOT_READY responses
990 uint8_t retry_count = 0;
991 while (true) {
992 TMC9660_LOG_DEBUG(*static_cast<Derived*>(this), 2, "SPI_TMCL",
993 "[TMCL TX 2 ] %02X %02X %02X %02X %02X %02X %02X %02X%s", tx_buf[0],
994 tx_buf[1], tx_buf[2], tx_buf[3], tx_buf[4], tx_buf[5], tx_buf[6], tx_buf[7],
995 retry_count > 0 ? " (retry)" : "");
996
997 if (!spiTransferTMCL(tx_buf, rx_buf))
998 return false;
999
1000 TMC9660_LOG_DEBUG(*static_cast<Derived*>(this), 2, "SPI_TMCL",
1001 "[TMCL RX 2 ] %02X %02X %02X %02X %02X %02X %02X %02X", rx_buf[0], rx_buf[1],
1002 rx_buf[2], rx_buf[3], rx_buf[4], rx_buf[5], rx_buf[6], rx_buf[7]);
1003
1004 // Check SPI status byte (first byte) for NOT_READY
1005 SPIStatus spi_status = static_cast<SPIStatus>(rx_buf[0]);
1006
1007 if (spi_status == SPIStatus::NOT_READY) {
1008 // System not ready - retry if we haven't exceeded max retries
1009 if (retry_count < this->spiRetryMaxCount_) {
1010 retry_count++;
1011 TMC9660_LOG_DEBUG(
1012 *static_cast<Derived*>(this), 2, "SPI_TMCL",
1013 "⚠️ SPI_STATUS_NOT_READY received, retrying (attempt %u/%u) after %u us", retry_count,
1014 this->spiRetryMaxCount_ + 1, this->spiRetryIntervalUs_);
1015 this->delayUs(this->spiRetryIntervalUs_);
1016 continue; // Retry the transaction
1017 } else {
1018 // Max retries exceeded - parse reply manually since fromSpi returns false for NOT_READY
1019 TMC9660_LOG_DEBUG(*static_cast<Derived*>(this), 1, "SPI_TMCL",
1020 "❌ SPI_STATUS_NOT_READY: Max retries (%u) exceeded",
1021 this->spiRetryMaxCount_);
1022 // Validate checksum before parsing
1023 if (tmclChecksum(rx_buf.data(), 7) != rx_buf[7]) {
1024 TMC9660_LOG_DEBUG(*static_cast<Derived*>(this), 1, "SPI_TMCL", "⚠️ Checksum error in NOT_READY response");
1025 return false;
1026 }
1027 // Manually populate reply structure for NOT_READY response
1028 std::copy(rx_buf.begin(), rx_buf.end(), reply.rawBytes.begin());
1029 reply.spi_status = SPIStatus::NOT_READY;
1030 reply.status = rx_buf[1]; // TMCL status
1031 reply.opcode = rx_buf[2]; // Operation code
1032 reply.value = (static_cast<uint32_t>(rx_buf[3]) << 24) |
1033 (static_cast<uint32_t>(rx_buf[4]) << 16) |
1034 (static_cast<uint32_t>(rx_buf[5]) << 8) | static_cast<uint32_t>(rx_buf[6]);
1035 return false; // Return false to indicate failure
1036 }
1037 }
1038
1039 // Not NOT_READY - parse and return normally
1040 break;
1041 }
1042
1043 // Parse reply with command context (for handling special reply formats like GetVersion string)
1044 bool parse_ok = TMCLReply::fromSpi(rx_buf, reply, tx.opcode, tx.type);
1045 if (!parse_ok && static_cast<SPIStatus>(rx_buf[0]) == SPIStatus::NOT_READY) {
1046 // This shouldn't happen due to retry logic, but handle it just in case
1047 TMC9660_LOG_DEBUG(*static_cast<Derived*>(this), 1, "SPI_TMCL", "⚠️ Unexpected NOT_READY after retry loop");
1048 }
1049 return parse_ok;
1050 }
