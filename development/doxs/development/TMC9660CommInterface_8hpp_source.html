252 [[nodiscard]] std::string getVersionString() const noexcept {
253 if (opcode != 136) return ""; // Not a GetVersion reply

283 static bool fromSpi(std::span<const uint8_t, 8> in, TMCLReply &r,
284 uint8_t sentOpcode = 0, uint16_t sentType = 0) noexcept {
285 // Store raw bytes for advanced parsing (ALWAYS, even if parsing fails)
286 std::copy(in.begin(), in.end(), r.rawBytes.begin());
287
288 // Check for SESSION_START status codes (bootloader or parameter mode)
289 uint8_t rawStatus = in[0];
290 if (rawStatus == 0x13) {
291 // SESSION_START from bootloader (0x13)
292 // These use different protocols, so TMCL checksum will fail
293 // But we still want to return success so caller can use rawBytes
294 r.spiStatus = static_cast<SPIStatus>(rawStatus);
295 r.status = in[1];
296 r.opcode = in[2];
297 r.value = (static_cast<uint32_t>(in[3]) << 24) | (static_cast<uint32_t>(in[4]) << 16) |
298 (static_cast<uint32_t>(in[5]) << 8) | static_cast<uint32_t>(in[6]);
299 return true; // Allow SESSION_START even if checksum fails
300 }
301
302 // Check for GetVersion string format (Command 136, Type 0)
303 // GetVersion with Type=0 returns a special string format without checksum
304 // Format: [Host Address][Version String 8 chars]
305 if (sentOpcode == 136 && sentType == 0) {
306 // We sent GetVersion Type=0, expect string format reply (no checksum validation)
307 r.spiStatus = SPIStatus::OK; // Assume OK for version string
308 r.status = 100; // REPLY_OK for successful version retrieval
309 r.opcode = 136; // GetVersion command
310 r.value = 0; // No value field for string format
311 return true; // Success - skip checksum validation for string format
312 }
313
314 // Standard TMCL reply validation
315 if (tmclChecksum(in.data(), 7) != in[7])
316 return false;
317 r.spiStatus = static_cast<SPIStatus>(in[0]);
318 if (r.spiStatus == SPIStatus::NOT_READY || r.spiStatus == SPIStatus::CHECKSUM_ERROR)
319 return false;
320 r.status = in[1];
321 r.opcode = in[2];
322 r.value = (static_cast<uint32_t>(in[3]) << 24) | (static_cast<uint32_t>(in[4]) << 16) |
323 (static_cast<uint32_t>(in[5]) << 8) | static_cast<uint32_t>(in[6]);
324 return true;
325 }

344 static bool fromUart(std::span<const uint8_t, 9> in, uint8_t addr, TMCLReply &r,
345 uint8_t sentOpcode = 0, uint16_t sentType = 0) noexcept {
346 // Store raw bytes (map 9-byte UART to 8-byte format, skip first byte)
347 std::copy(in.begin() + 1, in.end(), r.rawBytes.begin());
348
349 // UART Reply Format (per datasheet):
350 // byte0: Host Address (8 bits)
351 // byte1: Sync bit (bit 0) + Module Address (bits 1-7)
352 // byte2: TMCL Status
353 // byte3: Operation
354 // byte4-7: Data (big-endian)
355 // byte8: Checksum
356
357 // Check for GetVersion string format (Command 136, Type 0)
358 // GetVersion with Type=0 returns a special string format without valid checksum
359 // Format: [Host Address][Sync+Address][Version String 7 chars]
360 if (sentOpcode == 136 && sentType == 0) {
361 // We sent GetVersion Type=0, expect string format reply (no checksum validation)
362 r.spiStatus = SPIStatus::OK; // Assume OK for version string
363 r.status = 100; // REPLY_OK for successful version retrieval
364 r.opcode = 136; // GetVersion command
365 r.value = 0; // No value field for string format
366 return true; // Success - skip checksum validation for string format
367 }
368
369 // Verify module address in byte 1 (bits 1-7)
370 // "The module address reuses the upper 7 bits of the bootloader device address"
371 // Compare the upper 7 bits of byte 1 with the upper 7 bits of the address
372 if ((in[1] & 0xFE) != (addr & 0xFE))
373 return false;
374
375 // Verify checksum over first 8 bytes
376 if (tmclChecksum(in.data(), 8) != in[8])
377 return false;
378
379 r.spiStatus = SPIStatus::OK; // UART has no SPI status field
380 r.status = in[2]; // TMCL Status at byte 2
381 r.opcode = in[3]; // Operation at byte 3
382 r.value = (static_cast<uint32_t>(in[4]) << 24) | (static_cast<uint32_t>(in[5]) << 16) |
383 (static_cast<uint32_t>(in[6]) << 8) | static_cast<uint32_t>(in[7]);
384 return true;
385 }

403 void toSpi(std::span<uint8_t, 8> out) const noexcept {
404 out[0] = opcode;
405 out[1] = static_cast<uint8_t>(type & 0xFF); // Type lower 8 bits (bits 0-7)
406 // BYTE2: bits 20-23 (high nibble) = Motor/Bank, bits 16-19 (upper nibble) = Type upper 4 bits
407 out[2] = static_cast<uint8_t>(((motor & 0x0F) << 4) | ((type & 0xF00) >> 8));
408 out[3] = static_cast<uint8_t>(value >> 24);
409 out[4] = static_cast<uint8_t>(value >> 16);
410 out[5] = static_cast<uint8_t>(value >> 8);
411 out[6] = static_cast<uint8_t>(value);
412 out[7] = tmclChecksum(out.data(), 7);
413 }

420 void toUart(uint8_t addr, std::span<uint8_t, 9> out) const noexcept {
421 // Byte 0: [Sync bit (bit 0)] [Module Address (bits 1-7)]
422 // "The module address reuses the upper 7 bits of the bootloader device address"
423 // This means we take bits 7-1 of the address and place them in bits 7-1 of byte 0
424 // Then set bit 0 (sync bit) to 1
425 out[0] = (addr & 0xFE) | 0x01; // Keep upper 7 bits of address, set sync bit
426 out[1] = opcode;
427 out[2] = static_cast<uint8_t>(type & 0xFF); // Type field (8 bits)
428 out[3] = static_cast<uint8_t>(motor & 0x0F); // Motor/Bank field (4 bits)
429 out[4] = static_cast<uint8_t>(value >> 24);
430 out[5] = static_cast<uint8_t>(value >> 16);
431 out[6] = static_cast<uint8_t>(value >> 8);
432 out[7] = static_cast<uint8_t>(value);
433 out[8] = tmclChecksum(out.data(), 8);
434 }

441 static TMCLFrame fromSpi(std::span<const uint8_t, 8> in) noexcept {
442 TMCLFrame f;
443 f.opcode = in[0];
444 // BYTE1: Type lower 8 bits (bits 0-7)
445 // BYTE2: bits 20-23 (high nibble) = Motor/Bank, bits 16-19 (low nibble) = Type upper 4 bits
446 f.type = static_cast<uint16_t>(in[1] | ((in[2] & 0x0F) << 8));
447 f.motor = (in[2] >> 4) & 0x0F;
448 f.value = (static_cast<uint32_t>(in[3]) << 24) | (static_cast<uint32_t>(in[4]) << 16) |
449 (static_cast<uint32_t>(in[5]) << 8) | static_cast<uint32_t>(in[6]);
450 return f;
451 }

869 bool transferTMCL(const TMCLFrame &tx, TMCLReply &reply, uint8_t,
870 TMCLReply *firstReply, const TMCLFrame *secondCommand) noexcept override {
871 std::array<uint8_t, 8> txBuf, rxBuf;
872 tx.toSpi(txBuf);
873
874 // Log raw SPI bytes being transmitted
875 TMC9660_LOG_DEBUG(*this, 2, "SPI_TMCL", "[TMCL TX 1 ] %02X %02X %02X %02X %02X %02X %02X %02X",
876 txBuf[0], txBuf[1], txBuf[2], txBuf[3],
877 txBuf[4], txBuf[5], txBuf[6], txBuf[7]);
878
879 // Transaction 1: Send command, receive first reply
880 if (!spiTransferTMCL(txBuf, rxBuf))
881 return false;
882
883 TMC9660_LOG_DEBUG(*this, 2, "SPI_TMCL", "[TMCL RX 1 ] %02X %02X %02X %02X %02X %02X %02X %02X",
884 rxBuf[0], rxBuf[1], rxBuf[2], rxBuf[3],
885 rxBuf[4], rxBuf[5], rxBuf[6], rxBuf[7]);
886
887 // Optionally capture first reply (this is the reply to the PREVIOUS command due to SPI delay)
888 if (firstReply) {
889 bool firstParseOk = TMCLReply::fromSpi(rxBuf, *firstReply);
890 TMC9660_LOG_DEBUG(*this, 2, "SPI_TMCL", " └─> SPI_Status=0x%02X, TMCL_Status=0x%02X, Value=0x%08X (parse %s)",
891 static_cast<uint8_t>(firstReply->spiStatus), firstReply->status, firstReply->value,
892 firstParseOk ? "OK" : "failed");
893 // Note: We don't return false here because SESSION_START may have non-standard format
894 // The caller can check rawBytes[0] for SESSION_START status codes
895 }
896
897 // Transaction 2: Send second command (or NO_OP), receive final reply
898 // This transaction is wrapped in retry logic for SPI_STATUS_NOT_READY
899 TMCLFrame cmd2 = secondCommand ? *secondCommand : TMCLFrame{}; // NO_OP if not provided
900 cmd2.toSpi(txBuf);
901
902 // Retry loop for SPI_STATUS_NOT_READY responses
903 uint8_t retryCount = 0;
904 while (true) {
905 TMC9660_LOG_DEBUG(*this, 2, "SPI_TMCL", "[TMCL TX 2 ] %02X %02X %02X %02X %02X %02X %02X %02X%s",
906 txBuf[0], txBuf[1], txBuf[2], txBuf[3],
907 txBuf[4], txBuf[5], txBuf[6], txBuf[7],
908 retryCount > 0 ? " (retry)" : "");
909
910 if (!spiTransferTMCL(txBuf, rxBuf))
911 return false;
912
913 TMC9660_LOG_DEBUG(*this, 2, "SPI_TMCL", "[TMCL RX 2 ] %02X %02X %02X %02X %02X %02X %02X %02X",
914 rxBuf[0], rxBuf[1], rxBuf[2], rxBuf[3],
915 rxBuf[4], rxBuf[5], rxBuf[6], rxBuf[7]);
916
917 // Check SPI status byte (first byte) for NOT_READY
918 SPIStatus spiStatus = static_cast<SPIStatus>(rxBuf[0]);
919
920 if (spiStatus == SPIStatus::NOT_READY) {
921 // System not ready - retry if we haven't exceeded max retries
922 if (retryCount < spiRetryMaxCount_) {
923 retryCount++;
924 TMC9660_LOG_DEBUG(*this, 2, "SPI_TMCL", "⚠️ SPI_STATUS_NOT_READY received, retrying (attempt %u/%u) after %u us",
925 retryCount, spiRetryMaxCount_ + 1, spiRetryIntervalUs_);
926 delayUs(spiRetryIntervalUs_);
927 continue; // Retry the transaction
928 } else {
929 // Max retries exceeded - parse reply manually since fromSpi returns false for NOT_READY
930 TMC9660_LOG_DEBUG(*this, 1, "SPI_TMCL", "❌ SPI_STATUS_NOT_READY: Max retries (%u) exceeded", spiRetryMaxCount_);
931 // Validate checksum before parsing
932 if (tmclChecksum(rxBuf.data(), 7) != rxBuf[7]) {
933 TMC9660_LOG_DEBUG(*this, 1, "SPI_TMCL", "⚠️ Checksum error in NOT_READY response");
934 return false;
935 }
936 // Manually populate reply structure for NOT_READY response
937 std::copy(rxBuf.begin(), rxBuf.end(), reply.rawBytes.begin());
938 reply.spiStatus = SPIStatus::NOT_READY;
939 reply.status = rxBuf[1]; // TMCL status
940 reply.opcode = rxBuf[2]; // Operation code
941 reply.value = (static_cast<uint32_t>(rxBuf[3]) << 24) |
942 (static_cast<uint32_t>(rxBuf[4]) << 16) |
943 (static_cast<uint32_t>(rxBuf[5]) << 8) |
944 static_cast<uint32_t>(rxBuf[6]);
945 return false; // Return false to indicate failure
946 }
947 }
948
949 // Not NOT_READY - parse and return normally
950 break;
951 }
952
953 // Parse reply with command context (for handling special reply formats like GetVersion string)
954 bool parseOk = TMCLReply::fromSpi(rxBuf, reply, tx.opcode, tx.type);
955 if (!parseOk && static_cast<SPIStatus>(rxBuf[0]) == SPIStatus::NOT_READY) {
956 // This shouldn't happen due to retry logic, but handle it just in case
957 TMC9660_LOG_DEBUG(*this, 1, "SPI_TMCL", "⚠️ Unexpected NOT_READY after retry loop");
958 }
959 return parseOk;
960 }
