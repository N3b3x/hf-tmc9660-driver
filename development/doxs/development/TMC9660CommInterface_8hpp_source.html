240 [[nodiscard]] uint32_t extractRawValue(size_t offset) const noexcept {

242 return 0;

244 (static_cast<uint32_t>(rawBytes[offset + 1]) << 16) |

256 [[nodiscard]] std::string getVersionString() const noexcept {
257 if (opcode != 136)
258 return ""; // Not a GetVersion reply
259
260 std::string version;
261 // Extract version string from bytes 1-8 (skip host address at byte 0)
262 for (size_t i = 1; i < 8; ++i) {
263 if (rawBytes[i] >= 0x20 && rawBytes[i] <= 0x7E) { // Printable ASCII
264 version += static_cast<char>(rawBytes[i]);

266 break; // Stop at first non-printable character
267 }

288 static bool fromSpi(std::span<const uint8_t, 8> in, TMCLReply& r, uint8_t sentOpcode = 0,
289 uint16_t sentType = 0) noexcept {
290 // Store raw bytes for advanced parsing (ALWAYS, even if parsing fails)
291 std::copy(in.begin(), in.end(), r.rawBytes.begin());
292
293 // Check for SESSION_START status codes (bootloader or parameter mode)
294 uint8_t rawStatus = in[0];
295 if (rawStatus == 0x13) {
296 // SESSION_START from bootloader (0x13)
297 // These use different protocols, so TMCL checksum will fail
298 // But we still want to return success so caller can use rawBytes
299 r.spiStatus = static_cast<SPIStatus>(rawStatus);
300 r.status = in[1];
301 r.opcode = in[2];
302 r.value = (static_cast<uint32_t>(in[3]) << 24) | (static_cast<uint32_t>(in[4]) << 16) |
303 (static_cast<uint32_t>(in[5]) << 8) | static_cast<uint32_t>(in[6]);
304 return true; // Allow SESSION_START even if checksum fails
305 }
306
307 // Check for GetVersion string format (Command 136, Type 0)
308 // GetVersion with Type=0 returns a special string format without checksum
309 // Format: [Host Address][Version String 8 chars]
310 if (sentOpcode == 136 && sentType == 0) {
311 // We sent GetVersion Type=0, expect string format reply (no checksum validation)
312 r.spiStatus = SPIStatus::OK; // Assume OK for version string
313 r.status = 100; // REPLY_OK for successful version retrieval
314 r.opcode = 136; // GetVersion command
315 r.value = 0; // No value field for string format
316 return true; // Success - skip checksum validation for string format
317 }
318
319 // Standard TMCL reply validation
320 if (tmclChecksum(in.data(), 7) != in[7])
321 return false;
322 r.spiStatus = static_cast<SPIStatus>(in[0]);
323 if (r.spiStatus == SPIStatus::NOT_READY || r.spiStatus == SPIStatus::CHECKSUM_ERROR)
324 return false;
325 r.status = in[1];
326 r.opcode = in[2];
327 r.value = (static_cast<uint32_t>(in[3]) << 24) | (static_cast<uint32_t>(in[4]) << 16) |
328 (static_cast<uint32_t>(in[5]) << 8) | static_cast<uint32_t>(in[6]);
329 return true;
330 }

349 static bool fromUart(std::span<const uint8_t, 9> in, uint8_t addr, TMCLReply& r,
350 uint8_t sentOpcode = 0, uint16_t sentType = 0) noexcept {
351 // Store raw bytes (map 9-byte UART to 8-byte format, skip first byte)
352 std::copy(in.begin() + 1, in.end(), r.rawBytes.begin());
353
354 // UART Reply Format (per datasheet):
355 // byte0: Host Address (8 bits)
356 // byte1: Sync bit (bit 0) + Module Address (bits 1-7)
357 // byte2: TMCL Status
358 // byte3: Operation
359 // byte4-7: Data (big-endian)
360 // byte8: Checksum
361
362 // Check for GetVersion string format (Command 136, Type 0)
363 // GetVersion with Type=0 returns a special string format without valid checksum
364 // Format: [Host Address][Sync+Address][Version String 7 chars]
365 if (sentOpcode == 136 && sentType == 0) {
366 // We sent GetVersion Type=0, expect string format reply (no checksum validation)
367 r.spiStatus = SPIStatus::OK; // Assume OK for version string
368 r.status = 100; // REPLY_OK for successful version retrieval
369 r.opcode = 136; // GetVersion command
370 r.value = 0; // No value field for string format
371 return true; // Success - skip checksum validation for string format
372 }
373
374 // Verify module address in byte 1 (bits 1-7)
375 // "The module address reuses the upper 7 bits of the bootloader device address"
376 // Compare the upper 7 bits of byte 1 with the upper 7 bits of the address
377 if ((in[1] & 0xFE) != (addr & 0xFE))
378 return false;
379
380 // Verify checksum over first 8 bytes
381 if (tmclChecksum(in.data(), 8) != in[8])
382 return false;
383
384 r.spiStatus = SPIStatus::OK; // UART has no SPI status field
385 r.status = in[2]; // TMCL Status at byte 2
386 r.opcode = in[3]; // Operation at byte 3
387 r.value = (static_cast<uint32_t>(in[4]) << 24) | (static_cast<uint32_t>(in[5]) << 16) |
388 (static_cast<uint32_t>(in[6]) << 8) | static_cast<uint32_t>(in[7]);
389 return true;
390 }

398struct TMCLFrame {
399 uint8_t opcode = 0;
400 uint16_t type = 0;
401 uint8_t motor = 0;

408 void toSpi(std::span<uint8_t, 8> out) const noexcept {
409 out[0] = opcode;
410 out[1] = static_cast<uint8_t>(type & 0xFF); // Type lower 8 bits (bits 0-7)
411 // BYTE2: bits 20-23 (high nibble) = Motor/Bank, bits 16-19 (upper nibble) = Type upper 4 bits
412 out[2] = static_cast<uint8_t>(((motor & 0x0F) << 4) | ((type & 0xF00) >> 8));
413 out[3] = static_cast<uint8_t>(value >> 24);
414 out[4] = static_cast<uint8_t>(value >> 16);
415 out[5] = static_cast<uint8_t>(value >> 8);
416 out[6] = static_cast<uint8_t>(value);
417 out[7] = tmclChecksum(out.data(), 7);
418 }

425 void toUart(uint8_t addr, std::span<uint8_t, 9> out) const noexcept {
426 // Byte 0: [Sync bit (bit 0)] [Module Address (bits 1-7)]
427 // "The module address reuses the upper 7 bits of the bootloader device address"
428 // This means we take bits 7-1 of the address and place them in bits 7-1 of byte 0
429 // Then set bit 0 (sync bit) to 1
430 out[0] = (addr & 0xFE) | 0x01; // Keep upper 7 bits of address, set sync bit
431 out[1] = opcode;
432 out[2] = static_cast<uint8_t>(type & 0xFF); // Type field (8 bits)
433 out[3] = static_cast<uint8_t>(motor & 0x0F); // Motor/Bank field (4 bits)
434 out[4] = static_cast<uint8_t>(value >> 24);
435 out[5] = static_cast<uint8_t>(value >> 16);
436 out[6] = static_cast<uint8_t>(value >> 8);
437 out[7] = static_cast<uint8_t>(value);
438 out[8] = tmclChecksum(out.data(), 8);
439 }

446 static TMCLFrame fromSpi(std::span<const uint8_t, 8> in) noexcept {
447 TMCLFrame f;
448 f.opcode = in[0];
449 // BYTE1: Type lower 8 bits (bits 0-7)
450 // BYTE2: bits 20-23 (high nibble) = Motor/Bank, bits 16-19 (low nibble) = Type upper 4 bits
451 f.type = static_cast<uint16_t>(in[1] | ((in[2] & 0x0F) << 8));
452 f.motor = (in[2] >> 4) & 0x0F;
453 f.value = (static_cast<uint32_t>(in[3]) << 24) | (static_cast<uint32_t>(in[4]) << 16) |
454 (static_cast<uint32_t>(in[5]) << 8) | static_cast<uint32_t>(in[6]);
455 return f;
456 }

875 bool transferTMCL(const TMCLFrame& tx, TMCLReply& reply, uint8_t, TMCLReply* firstReply,
876 const TMCLFrame* secondCommand) noexcept override {
877 std::array<uint8_t, 8> txBuf, rxBuf;
878 tx.toSpi(txBuf);
879
880 // Log raw SPI bytes being transmitted
881 TMC9660_LOG_DEBUG(*this, 2, "SPI_TMCL", "[TMCL TX 1 ] %02X %02X %02X %02X %02X %02X %02X %02X",
882 txBuf[0], txBuf[1], txBuf[2], txBuf[3], txBuf[4], txBuf[5], txBuf[6],
883 txBuf[7]);
884
885 // Transaction 1: Send command, receive first reply
886 if (!spiTransferTMCL(txBuf, rxBuf))
887 return false;
888
889 TMC9660_LOG_DEBUG(*this, 2, "SPI_TMCL", "[TMCL RX 1 ] %02X %02X %02X %02X %02X %02X %02X %02X",
890 rxBuf[0], rxBuf[1], rxBuf[2], rxBuf[3], rxBuf[4], rxBuf[5], rxBuf[6],
891 rxBuf[7]);
892
893 // Optionally capture first reply (this is the reply to the PREVIOUS command due to SPI delay)
894 if (firstReply) {
895 bool firstParseOk = TMCLReply::fromSpi(rxBuf, *firstReply);
896 TMC9660_LOG_DEBUG(
897 *this, 2, "SPI_TMCL",
898 " └─> SPI_Status=0x%02X, TMCL_Status=0x%02X, Value=0x%08X (parse %s)",
899 static_cast<uint8_t>(firstReply->spiStatus), firstReply->status, firstReply->value,
900 firstParseOk ? "OK" : "failed");
901 // Note: We don't return false here because SESSION_START may have non-standard format
902 // The caller can check rawBytes[0] for SESSION_START status codes
903 }
904
905 // Transaction 2: Send second command (or NO_OP), receive final reply
906 // This transaction is wrapped in retry logic for SPI_STATUS_NOT_READY
907 TMCLFrame cmd2 = secondCommand ? *secondCommand : TMCLFrame{}; // NO_OP if not provided
908 cmd2.toSpi(txBuf);
909
910 // Retry loop for SPI_STATUS_NOT_READY responses
911 uint8_t retryCount = 0;
912 while (true) {
913 TMC9660_LOG_DEBUG(*this, 2, "SPI_TMCL",
914 "[TMCL TX 2 ] %02X %02X %02X %02X %02X %02X %02X %02X%s", txBuf[0],
915 txBuf[1], txBuf[2], txBuf[3], txBuf[4], txBuf[5], txBuf[6], txBuf[7],
916 retryCount > 0 ? " (retry)" : "");
917
918 if (!spiTransferTMCL(txBuf, rxBuf))
919 return false;
920
921 TMC9660_LOG_DEBUG(*this, 2, "SPI_TMCL",
922 "[TMCL RX 2 ] %02X %02X %02X %02X %02X %02X %02X %02X", rxBuf[0], rxBuf[1],
923 rxBuf[2], rxBuf[3], rxBuf[4], rxBuf[5], rxBuf[6], rxBuf[7]);
924
925 // Check SPI status byte (first byte) for NOT_READY
926 SPIStatus spiStatus = static_cast<SPIStatus>(rxBuf[0]);
927
928 if (spiStatus == SPIStatus::NOT_READY) {
929 // System not ready - retry if we haven't exceeded max retries
930 if (retryCount < spiRetryMaxCount_) {
931 retryCount++;
932 TMC9660_LOG_DEBUG(
933 *this, 2, "SPI_TMCL",
934 "⚠️ SPI_STATUS_NOT_READY received, retrying (attempt %u/%u) after %u us", retryCount,
935 spiRetryMaxCount_ + 1, spiRetryIntervalUs_);
936 delayUs(spiRetryIntervalUs_);
937 continue; // Retry the transaction
938 } else {
939 // Max retries exceeded - parse reply manually since fromSpi returns false for NOT_READY
940 TMC9660_LOG_DEBUG(*this, 1, "SPI_TMCL",
941 "❌ SPI_STATUS_NOT_READY: Max retries (%u) exceeded",
942 spiRetryMaxCount_);
943 // Validate checksum before parsing
944 if (tmclChecksum(rxBuf.data(), 7) != rxBuf[7]) {
945 TMC9660_LOG_DEBUG(*this, 1, "SPI_TMCL", "⚠️ Checksum error in NOT_READY response");
946 return false;
947 }
948 // Manually populate reply structure for NOT_READY response
949 std::copy(rxBuf.begin(), rxBuf.end(), reply.rawBytes.begin());
950 reply.spiStatus = SPIStatus::NOT_READY;
951 reply.status = rxBuf[1]; // TMCL status
952 reply.opcode = rxBuf[2]; // Operation code
953 reply.value = (static_cast<uint32_t>(rxBuf[3]) << 24) |
954 (static_cast<uint32_t>(rxBuf[4]) << 16) |
955 (static_cast<uint32_t>(rxBuf[5]) << 8) | static_cast<uint32_t>(rxBuf[6]);
956 return false; // Return false to indicate failure
957 }
958 }
959
960 // Not NOT_READY - parse and return normally
961 break;
962 }
963
964 // Parse reply with command context (for handling special reply formats like GetVersion string)
965 bool parseOk = TMCLReply::fromSpi(rxBuf, reply, tx.opcode, tx.type);
966 if (!parseOk && static_cast<SPIStatus>(rxBuf[0]) == SPIStatus::NOT_READY) {
967 // This shouldn't happen due to retry logic, but handle it just in case
968 TMC9660_LOG_DEBUG(*this, 1, "SPI_TMCL", "⚠️ Unexpected NOT_READY after retry loop");
969 }
970 return parseOk;
971 }
