284 static bool fromSpi(std::span<const uint8_t, 8> in, TMCLReply& r, uint8_t sent_opcode = 0,
285 uint16_t sent_type = 0) noexcept {
286 // Store raw bytes for advanced parsing (ALWAYS, even if parsing fails)
287 std::copy(in.begin(), in.end(), r.rawBytes.begin());
288
289 // Check for SESSION_START status codes (bootloader or parameter mode)
290 uint8_t raw_status = in[0];
291 if (raw_status == 0x13) {
292 // SESSION_START from bootloader (0x13)
293 // These use different protocols, so TMCL checksum will fail
294 // But we still want to return success so caller can use rawBytes
295 r.spi_status = static_cast<SPIStatus>(raw_status);
296 r.status = in[1];
297 r.opcode = in[2];
298 r.value = (static_cast<uint32_t>(in[3]) << 24) | (static_cast<uint32_t>(in[4]) << 16) |
299 (static_cast<uint32_t>(in[5]) << 8) | static_cast<uint32_t>(in[6]);
300 return true; // Allow SESSION_START even if checksum fails
301 }
302
303 // Check for GetVersion string format (Command 136, Type 0)
304 // GetVersion with Type=0 returns a special string format without checksum
305 // Format: [Host Address][Version String 8 chars]
306 if (sent_opcode == 136 && sent_type == 0) {
307 // We sent GetVersion Type=0, expect string format reply (no checksum validation)
308 r.spi_status = SPIStatus::OK; // Assume OK for version string
309 r.status = 100; // REPLY_OK for successful version retrieval
310 r.opcode = 136; // GetVersion command
311 r.value = 0; // No value field for string format
312 return true; // Success - skip checksum validation for string format
313 }
314
315 // Standard TMCL reply validation
316 if (tmclChecksum(in.data(), 7) != in[7])
317 return false;
318 r.spi_status = static_cast<SPIStatus>(in[0]);
319 if (r.spi_status == SPIStatus::NOT_READY || r.spi_status == SPIStatus::CHECKSUM_ERROR)
320 return false;
321 r.status = in[1];
322 r.opcode = in[2];
323 r.value = (static_cast<uint32_t>(in[3]) << 24) | (static_cast<uint32_t>(in[4]) << 16) |
324 (static_cast<uint32_t>(in[5]) << 8) | static_cast<uint32_t>(in[6]);
325 return true;
326 }

345 static bool fromUart(std::span<const uint8_t, 9> in, uint8_t addr, TMCLReply& r,
346 uint8_t sent_opcode = 0, uint16_t sent_type = 0) noexcept {
347 // Store raw bytes (map 9-byte UART to 8-byte format, skip first byte)
348 std::copy(in.begin() + 1, in.end(), r.rawBytes.begin());
349
350 // UART Reply Format (per datasheet):
351 // byte0: Host Address (8 bits)
352 // byte1: Sync bit (bit 0) + Module Address (bits 1-7)
353 // byte2: TMCL Status
354 // byte3: Operation
355 // byte4-7: Data (big-endian)
356 // byte8: Checksum
357
358 // Check for GetVersion string format (Command 136, Type 0)
359 // GetVersion with Type=0 returns a special string format without valid checksum
360 // Format: [Host Address][Sync+Address][Version String 7 chars]
361 if (sent_opcode == 136 && sent_type == 0) {
362 // We sent GetVersion Type=0, expect string format reply (no checksum validation)
363 r.spi_status = SPIStatus::OK; // Assume OK for version string
364 r.status = 100; // REPLY_OK for successful version retrieval
365 r.opcode = 136; // GetVersion command
366 r.value = 0; // No value field for string format
367 return true; // Success - skip checksum validation for string format
368 }
369
370 // Verify module address in byte 1 (bits 1-7)
371 // "The module address reuses the upper 7 bits of the bootloader device address"
372 // Compare the upper 7 bits of byte 1 with the upper 7 bits of the address
373 if ((in[1] & 0xFE) != (addr & 0xFE))
374 return false;
375
376 // Verify checksum over first 8 bytes
377 if (tmclChecksum(in.data(), 8) != in[8])
378 return false;
379
380 r.spi_status = SPIStatus::OK; // UART has no SPI status field
381 r.status = in[2]; // TMCL Status at byte 2
382 r.opcode = in[3]; // Operation at byte 3
383 r.value = (static_cast<uint32_t>(in[4]) << 24) | (static_cast<uint32_t>(in[5]) << 16) |
384 (static_cast<uint32_t>(in[6]) << 8) | static_cast<uint32_t>(in[7]);
385 return true;
386 }

411 void toSpi(std::span<uint8_t, 8> out) const noexcept {
412 out[0] = opcode;
413 out[1] = static_cast<uint8_t>(type & 0xFF);
414 out[2] = static_cast<uint8_t>(((type & 0xF00u) >> 4) | (motor & 0x0Fu));
415 out[3] = static_cast<uint8_t>(value >> 24);
416 out[4] = static_cast<uint8_t>(value >> 16);
417 out[5] = static_cast<uint8_t>(value >> 8);
418 out[6] = static_cast<uint8_t>(value);
419 out[7] = tmclChecksum(out.data(), 7);
420 }

427 void toUart(uint8_t addr, std::span<uint8_t, 9> out) const noexcept {
428 // Byte 0: [Sync bit (bit 0)] [Module Address (bits 1-7)]
429 // "The module address reuses the upper 7 bits of the bootloader device address"
430 // This means we take bits 7-1 of the address and place them in bits 7-1 of byte 0
431 // Then set bit 0 (sync bit) to 1
432 out[0] = (addr & 0xFE) | 0x01; // Keep upper 7 bits of address, set sync bit
433 out[1] = opcode;
434 // Bytes 2-3: 12-bit TYPE field + 4-bit MOTOR/BANK field, packed exactly like the
435 // Trinamic reference (`examples/TMC9660/TMC9660.c::sendRequestUART`):
436 // data[2] = type & 0xFF; // type[7:0]
437 // data[3] = (type >> 8) << 4 | (index & 0x0F); // type[11:8] hi-nibble | motor lo-nibble
438 // The previous packing dropped type[11:8] entirely and put motor in the low
439 // nibble, which silently aliased every parameter NR >= 256 down into the
440 // 0..255 range. That made the gate-driver overcurrent / VGS / supply / temp
441 // SAPs (NR 254..299) write the wrong target on the chip, and made the read
442 // scan return totally bogus values for NRs 256..334.
443 out[2] = static_cast<uint8_t>(type & 0xFFu);
444 out[3] = static_cast<uint8_t>(((type & 0xF00u) >> 4) | (motor & 0x0Fu));
445 out[4] = static_cast<uint8_t>(value >> 24);
446 out[5] = static_cast<uint8_t>(value >> 16);
447 out[6] = static_cast<uint8_t>(value >> 8);
448 out[7] = static_cast<uint8_t>(value);
449 out[8] = tmclChecksum(out.data(), 8);
450 }

989 TMCLReply* first_reply, const TMCLFrame* second_command) noexcept {
990 std::array<uint8_t, 8> tx_buf, rx_buf;
991 std::array<uint8_t, 8> tx2_buf, rx1_buf;
992 tx.toSpi(tx_buf);
993
994 TMC9660_LOG_TMCL_RAW_FRAME(*static_cast<Derived*>(this), 2, "SPI_TMCL", "[TMCL TX 1 ] %02X %02X %02X %02X %02X %02X %02X %02X",
995 tx_buf[0], tx_buf[1], tx_buf[2], tx_buf[3], tx_buf[4], tx_buf[5], tx_buf[6],
996 tx_buf[7]);
997
998 /* Opcode 0xFF, not 0x00 — see the @warning on this method. */
999 TMCLFrame cmd2{};
1000 cmd2.opcode = 0xFFu;
1001 if (second_command != nullptr) {
1002 cmd2 = *second_command;
1003 }
1004 cmd2.toSpi(tx2_buf);
1005
1006 /* Prefer hostSpiTransferTMCLPair when Derived provides it (STM32 shared-bus
1007 * TransferChain). Gap must be long enough for the TMCL parser to latch cmd1
1008 * before cmd2 clocks the reply — 300 µs was marginal at 1 MHz on Mid. */
1009 constexpr uint32_t kTmclPairGapUs = 1000U;
1010 bool pair_ok = false;
1011 if constexpr (requires(Derived& d, std::array<uint8_t, 8>& a, uint32_t g) {
1012 d.hostSpiTransferTMCLPair(a, a, a, a, g);
1013 }) {
1014 pair_ok = static_cast<Derived*>(this)->hostSpiTransferTMCLPair(
1015 tx_buf, rx1_buf, tx2_buf, rx_buf, kTmclPairGapUs);
1016 } else {

1018 return false;
1019 }

1021 pair_ok = spiTransferTMCL(tx2_buf, rx_buf);
1022 }
1023 if (!pair_ok) {
1024 return false;
1025 }

1029 rx1_buf[6], rx1_buf[7]);
1030 TMC9660_LOG_TMCL_RAW_FRAME(*static_cast<Derived*>(this), 2, "SPI_TMCL", "[TMCL TX 2 ] %02X %02X %02X %02X %02X %02X %02X %02X",
1031 tx2_buf[0], tx2_buf[1], tx2_buf[2], tx2_buf[3], tx2_buf[4], tx2_buf[5],
1032 tx2_buf[6], tx2_buf[7]);
1033 TMC9660_LOG_TMCL_RAW_FRAME(*static_cast<Derived*>(this), 2, "SPI_TMCL", "[TMCL RX 2 ] %02X %02X %02X %02X %02X %02X %02X %02X",
1034 rx_buf[0], rx_buf[1], rx_buf[2], rx_buf[3], rx_buf[4], rx_buf[5], rx_buf[6],
1035 rx_buf[7]);
1036
1037 // Optionally capture first reply (this is the reply to the PREVIOUS command due to SPI delay)
1038 if (first_reply) {
1039 const bool first_parse_ok = TMCLReply::fromSpi(rx1_buf, *first_reply);
1040 (void)first_parse_ok; // Only consumed by raw-frame logging, which is compiled out by default.
1041 TMC9660_LOG_TMCL_RAW_FRAME(
1042 *static_cast<Derived*>(this), 2, "SPI_TMCL",
1043 " └─> SPI_Status=0x%02X, TMCL_Status=0x%02X, Value=0x%08X (parse %s)",
1044 static_cast<uint8_t>(first_reply->spi_status), first_reply->status, first_reply->value,
1045 first_parse_ok ? "OK" : "failed");
1046 }

1048 // Two SPI statuses mean "reply not available yet" and are retryable by re-sending the
1049 // reply-request frame only (never the command, so a write cannot be duplicated):
1050 // 0xF0 NOT_READY — slave explicitly reports busy.
1051 // 0x00 CHECKSUM_ERROR — also the value seen for an all-zero read, which is what the
1052 // firmware-serviced slave streams while the FOC engine has not
1053 // queued the reply (GIM4305 bench: zeros under load, then valid).
1054 tx_buf = tx2_buf;
1055 uint8_t retry_count = 0;
1056 while (true) {
1057 SPIStatus spi_status = static_cast<SPIStatus>(rx_buf[0]);
1058
1059 if (spi_status == SPIStatus::NOT_READY ||
1060 spi_status == SPIStatus::CHECKSUM_ERROR) {
1061 if (retry_count < this->spiRetryMaxCount_) {
1062 retry_count++;
1063 TMC9660_LOG_TMCL_RAW_FRAME(
1064 *static_cast<Derived*>(this), 2, "SPI_TMCL",
1065 "⚠️ Reply not ready (SPI status 0x%02X), retrying (attempt %u/%u) after %u us",
1066 static_cast<unsigned>(rx_buf[0]), retry_count, this->spiRetryMaxCount_ + 1,
1067 this->spiRetryIntervalUs_);
1068 this->delayUs(this->spiRetryIntervalUs_);
1069 TMC9660_LOG_TMCL_RAW_FRAME(*static_cast<Derived*>(this), 2, "SPI_TMCL",
1070 "[TMCL TX 2 ] %02X %02X %02X %02X %02X %02X %02X %02X (retry)",
1071 tx_buf[0], tx_buf[1], tx_buf[2], tx_buf[3], tx_buf[4], tx_buf[5],
1072 tx_buf[6], tx_buf[7]);
1073 if (!spiTransferTMCL(tx_buf, rx_buf))
1074 return false;
1075 TMC9660_LOG_TMCL_RAW_FRAME(*static_cast<Derived*>(this), 2, "SPI_TMCL",
1076 "[TMCL RX 2 ] %02X %02X %02X %02X %02X %02X %02X %02X", rx_buf[0],
1077 rx_buf[1], rx_buf[2], rx_buf[3], rx_buf[4], rx_buf[5], rx_buf[6],
1078 rx_buf[7]);
1079 continue;
1080 } else {
1081 TMC9660_LOG_DEBUG(*static_cast<Derived*>(this), 1, "SPI_TMCL",
1082 "❌ SPI_STATUS_NOT_READY: Max retries (%u) exceeded",
1083 this->spiRetryMaxCount_);
1084 if (tmclChecksum(rx_buf.data(), 7) != rx_buf[7]) {
1085 TMC9660_LOG_DEBUG(*static_cast<Derived*>(this), 1, "SPI_TMCL", "⚠️ Checksum error in NOT_READY response");
1086 return false;
1087 }
1088 std::copy(rx_buf.begin(), rx_buf.end(), reply.rawBytes.begin());
1089 reply.spi_status = SPIStatus::NOT_READY;
1090 reply.status = rx_buf[1];
1091 reply.opcode = rx_buf[2];
1092 reply.value = (static_cast<uint32_t>(rx_buf[3]) << 24) |
1093 (static_cast<uint32_t>(rx_buf[4]) << 16) |
1094 (static_cast<uint32_t>(rx_buf[5]) << 8) | static_cast<uint32_t>(rx_buf[6]);
1095 return false;
1096 }
1097 }
1098
1099 break;
1100 }
