280 static bool fromSpi(std::span<const uint8_t, 8> in, TMCLReply& r, uint8_t sent_opcode = 0,
281 uint16_t sent_type = 0) noexcept {
282 // Store raw bytes for advanced parsing (ALWAYS, even if parsing fails)
283 std::copy(in.begin(), in.end(), r.rawBytes.begin());
284
285 // Check for SESSION_START status codes (bootloader or parameter mode)
286 uint8_t raw_status = in[0];
287 if (raw_status == 0x13) {
288 // SESSION_START from bootloader (0x13)
289 // These use different protocols, so TMCL checksum will fail
290 // But we still want to return success so caller can use rawBytes
291 r.spi_status = static_cast<SPIStatus>(raw_status);
292 r.status = in[1];
293 r.opcode = in[2];
294 r.value = (static_cast<uint32_t>(in[3]) << 24) | (static_cast<uint32_t>(in[4]) << 16) |
295 (static_cast<uint32_t>(in[5]) << 8) | static_cast<uint32_t>(in[6]);
296 return true; // Allow SESSION_START even if checksum fails
297 }
298
299 // Check for GetVersion string format (Command 136, Type 0)
300 // GetVersion with Type=0 returns a special string format without checksum
301 // Format: [Host Address][Version String 8 chars]
302 if (sent_opcode == 136 && sent_type == 0) {
303 // We sent GetVersion Type=0, expect string format reply (no checksum validation)
304 r.spi_status = SPIStatus::OK; // Assume OK for version string
305 r.status = 100; // REPLY_OK for successful version retrieval
306 r.opcode = 136; // GetVersion command
307 r.value = 0; // No value field for string format
308 return true; // Success - skip checksum validation for string format
309 }
310
311 // Standard TMCL reply validation
312 if (tmclChecksum(in.data(), 7) != in[7])
313 return false;
314 r.spi_status = static_cast<SPIStatus>(in[0]);
315 if (r.spi_status == SPIStatus::NOT_READY || r.spi_status == SPIStatus::CHECKSUM_ERROR)
316 return false;
317 r.status = in[1];
318 r.opcode = in[2];
319 r.value = (static_cast<uint32_t>(in[3]) << 24) | (static_cast<uint32_t>(in[4]) << 16) |
320 (static_cast<uint32_t>(in[5]) << 8) | static_cast<uint32_t>(in[6]);
321 return true;
322 }

341 static bool fromUart(std::span<const uint8_t, 9> in, uint8_t addr, TMCLReply& r,
342 uint8_t sent_opcode = 0, uint16_t sent_type = 0) noexcept {
343 // Store raw bytes (map 9-byte UART to 8-byte format, skip first byte)
344 std::copy(in.begin() + 1, in.end(), r.rawBytes.begin());
345
346 // UART Reply Format (per datasheet):
347 // byte0: Host Address (8 bits)
348 // byte1: Sync bit (bit 0) + Module Address (bits 1-7)
349 // byte2: TMCL Status
350 // byte3: Operation
351 // byte4-7: Data (big-endian)
352 // byte8: Checksum
353
354 // Check for GetVersion string format (Command 136, Type 0)
355 // GetVersion with Type=0 returns a special string format without valid checksum
356 // Format: [Host Address][Sync+Address][Version String 7 chars]
357 if (sent_opcode == 136 && sent_type == 0) {
358 // We sent GetVersion Type=0, expect string format reply (no checksum validation)
359 r.spi_status = SPIStatus::OK; // Assume OK for version string
360 r.status = 100; // REPLY_OK for successful version retrieval
361 r.opcode = 136; // GetVersion command
362 r.value = 0; // No value field for string format
363 return true; // Success - skip checksum validation for string format
364 }
365
366 // Verify module address in byte 1 (bits 1-7)
367 // "The module address reuses the upper 7 bits of the bootloader device address"
368 // Compare the upper 7 bits of byte 1 with the upper 7 bits of the address
369 if ((in[1] & 0xFE) != (addr & 0xFE))
370 return false;
371
372 // Verify checksum over first 8 bytes
373 if (tmclChecksum(in.data(), 8) != in[8])
374 return false;
375
376 r.spi_status = SPIStatus::OK; // UART has no SPI status field
377 r.status = in[2]; // TMCL Status at byte 2
378 r.opcode = in[3]; // Operation at byte 3
379 r.value = (static_cast<uint32_t>(in[4]) << 24) | (static_cast<uint32_t>(in[5]) << 16) |
380 (static_cast<uint32_t>(in[6]) << 8) | static_cast<uint32_t>(in[7]);
381 return true;
382 }

407 void toSpi(std::span<uint8_t, 8> out) const noexcept {
408 out[0] = opcode;
409 out[1] = static_cast<uint8_t>(type & 0xFF);
410 out[2] = static_cast<uint8_t>(((type & 0xF00u) >> 4) | (motor & 0x0Fu));
411 out[3] = static_cast<uint8_t>(value >> 24);
412 out[4] = static_cast<uint8_t>(value >> 16);
413 out[5] = static_cast<uint8_t>(value >> 8);
414 out[6] = static_cast<uint8_t>(value);
415 out[7] = tmclChecksum(out.data(), 7);
416 }

423 void toUart(uint8_t addr, std::span<uint8_t, 9> out) const noexcept {
424 // Byte 0: [Sync bit (bit 0)] [Module Address (bits 1-7)]
425 // "The module address reuses the upper 7 bits of the bootloader device address"
426 // This means we take bits 7-1 of the address and place them in bits 7-1 of byte 0
427 // Then set bit 0 (sync bit) to 1
428 out[0] = (addr & 0xFE) | 0x01; // Keep upper 7 bits of address, set sync bit
429 out[1] = opcode;
430 // Bytes 2-3: 12-bit TYPE field + 4-bit MOTOR/BANK field, packed exactly like the
431 // Trinamic reference (`examples/TMC9660/TMC9660.c::sendRequestUART`):
432 // data[2] = type & 0xFF; // type[7:0]
433 // data[3] = (type >> 8) << 4 | (index & 0x0F); // type[11:8] hi-nibble | motor lo-nibble
434 // The previous packing dropped type[11:8] entirely and put motor in the low
435 // nibble, which silently aliased every parameter NR >= 256 down into the
436 // 0..255 range. That made the gate-driver overcurrent / VGS / supply / temp
437 // SAPs (NR 254..299) write the wrong target on the chip, and made the read
438 // scan return totally bogus values for NRs 256..334.
439 out[2] = static_cast<uint8_t>(type & 0xFFu);
440 out[3] = static_cast<uint8_t>(((type & 0xF00u) >> 4) | (motor & 0x0Fu));
441 out[4] = static_cast<uint8_t>(value >> 24);
442 out[5] = static_cast<uint8_t>(value >> 16);
443 out[6] = static_cast<uint8_t>(value >> 8);
444 out[7] = static_cast<uint8_t>(value);
445 out[8] = tmclChecksum(out.data(), 8);
446 }

942 bool transferTMCL(const TMCLFrame& tx, TMCLReply& reply, uint8_t, TMCLReply* first_reply,
943 const TMCLFrame* second_command) noexcept {
944 std::array<uint8_t, 8> tx_buf, rx_buf;
945 tx.toSpi(tx_buf);
946
947 TMC9660_LOG_TMCL_RAW_FRAME(*static_cast<Derived*>(this), 2, "SPI_TMCL", "[TMCL TX 1 ] %02X %02X %02X %02X %02X %02X %02X %02X",
948 tx_buf[0], tx_buf[1], tx_buf[2], tx_buf[3], tx_buf[4], tx_buf[5], tx_buf[6],
949 tx_buf[7]);
950
951 // Transaction 1: Send command, receive first reply
952 if (!spiTransferTMCL(tx_buf, rx_buf))
953 return false;
954
955 TMC9660_LOG_TMCL_RAW_FRAME(*static_cast<Derived*>(this), 2, "SPI_TMCL", "[TMCL RX 1 ] %02X %02X %02X %02X %02X %02X %02X %02X",
956 rx_buf[0], rx_buf[1], rx_buf[2], rx_buf[3], rx_buf[4], rx_buf[5], rx_buf[6],
957 rx_buf[7]);
958
959 // Optionally capture first reply (this is the reply to the PREVIOUS command due to SPI delay)
960 if (first_reply) {
961 bool first_parse_ok = TMCLReply::fromSpi(rx_buf, *first_reply);
962 TMC9660_LOG_TMCL_RAW_FRAME(
963 *static_cast<Derived*>(this), 2, "SPI_TMCL",
964 " └─> SPI_Status=0x%02X, TMCL_Status=0x%02X, Value=0x%08X (parse %s)",
965 static_cast<uint8_t>(first_reply->spi_status), first_reply->status, first_reply->value,
966 first_parse_ok ? "OK" : "failed");
967 // Note: We don't return false here because SESSION_START may have non-standard format
968 // The caller can check rawBytes[0] for SESSION_START status codes
969 }
970
971 // Give the TMCL parser time to latch the command from transaction 1 before transaction 2
972 // clocks out its reply. Without this gap, some boards at >=1 MHz SPI return
973 // REPLY_INVALID_CMD / REPLY_INVALID_VALUE on the *second* transfer even though the
974 // wire bytes look valid (see HalSpiTmc9660Comm notes in hf-core).
975 static_cast<Derived*>(this)->delayUs(300);
976
977 // Transaction 2: Send second command (or NO_OP), receive final reply
978 // This transaction is wrapped in retry logic for SPI_STATUS_NOT_READY
979 TMCLFrame cmd2 = second_command ? *second_command : TMCLFrame{}; // NO_OP if not provided
980 cmd2.toSpi(tx_buf);
981
982 // Retry loop for SPI_STATUS_NOT_READY responses
983 uint8_t retry_count = 0;
984 while (true) {
985 TMC9660_LOG_TMCL_RAW_FRAME(*static_cast<Derived*>(this), 2, "SPI_TMCL",
986 "[TMCL TX 2 ] %02X %02X %02X %02X %02X %02X %02X %02X%s", tx_buf[0],
987 tx_buf[1], tx_buf[2], tx_buf[3], tx_buf[4], tx_buf[5], tx_buf[6], tx_buf[7],
988 retry_count > 0 ? " (retry)" : "");
989
990 if (!spiTransferTMCL(tx_buf, rx_buf))
991 return false;
992
993 TMC9660_LOG_TMCL_RAW_FRAME(*static_cast<Derived*>(this), 2, "SPI_TMCL",
994 "[TMCL RX 2 ] %02X %02X %02X %02X %02X %02X %02X %02X", rx_buf[0], rx_buf[1],
995 rx_buf[2], rx_buf[3], rx_buf[4], rx_buf[5], rx_buf[6], rx_buf[7]);
996
997 // Check SPI status byte (first byte) for NOT_READY
998 SPIStatus spi_status = static_cast<SPIStatus>(rx_buf[0]);
999
1000 if (spi_status == SPIStatus::NOT_READY) {
1001 // System not ready - retry if we haven't exceeded max retries
1002 if (retry_count < this->spiRetryMaxCount_) {
1003 retry_count++;
1004 TMC9660_LOG_TMCL_RAW_FRAME(
1005 *static_cast<Derived*>(this), 2, "SPI_TMCL",
1006 "⚠️ SPI_STATUS_NOT_READY received, retrying (attempt %u/%u) after %u us", retry_count,
1007 this->spiRetryMaxCount_ + 1, this->spiRetryIntervalUs_);
1008 this->delayUs(this->spiRetryIntervalUs_);
1009 continue; // Retry the transaction
1010 } else {
1011 // Max retries exceeded - parse reply manually since fromSpi returns false for NOT_READY
1012 TMC9660_LOG_DEBUG(*static_cast<Derived*>(this), 1, "SPI_TMCL",
1013 "❌ SPI_STATUS_NOT_READY: Max retries (%u) exceeded",
1014 this->spiRetryMaxCount_);
1015 // Validate checksum before parsing
1016 if (tmclChecksum(rx_buf.data(), 7) != rx_buf[7]) {
1017 TMC9660_LOG_DEBUG(*static_cast<Derived*>(this), 1, "SPI_TMCL", "⚠️ Checksum error in NOT_READY response");
1018 return false;
1019 }
1020 // Manually populate reply structure for NOT_READY response
1021 std::copy(rx_buf.begin(), rx_buf.end(), reply.rawBytes.begin());
1022 reply.spi_status = SPIStatus::NOT_READY;
1023 reply.status = rx_buf[1];
1024 reply.opcode = rx_buf[2];
1025 reply.value = (static_cast<uint32_t>(rx_buf[3]) << 24) |
1026 (static_cast<uint32_t>(rx_buf[4]) << 16) |
1027 (static_cast<uint32_t>(rx_buf[5]) << 8) | static_cast<uint32_t>(rx_buf[6]);
1028 return false; // Return false to indicate failure
1029 }
1030 }
1031
1032 // Not NOT_READY - parse and return normally
1033 break;
1034 }
1035
1036 // Parse reply with command context (for handling special reply formats like GetVersion string)
1037 bool parse_ok = TMCLReply::fromSpi(rx_buf, reply, tx.opcode, tx.type);
1038 if (!parse_ok && static_cast<SPIStatus>(rx_buf[0]) == SPIStatus::NOT_READY) {
1039 // This shouldn't happen due to retry logic, but handle it just in case
1040 TMC9660_LOG_DEBUG(*static_cast<Derived*>(this), 1, "SPI_TMCL", "⚠️ Unexpected NOT_READY after retry loop");
1041 }
1042 return parse_ok;
1043 }
